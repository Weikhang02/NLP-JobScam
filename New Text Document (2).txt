# ── Model loading ─────────────────────────────────────────────────────────────
MODEL_PATH = "./model"

@st.cache_resource
def load_model():
    tokenizer = AutoTokenizer.from_pretrained(MODEL_PATH)
    model = AutoModelForSequenceClassification.from_pretrained(MODEL_PATH)
    model.eval()
    return tokenizer, model

tokenizer, model = load_model()


# ── Helpers ───────────────────────────────────────────────────────────────────
def clean_text(text: str) -> str:
    text = text.lower()
    text = re.sub(r'\n', ' ', text)
    text = re.sub(r'\s+', ' ', text)
    text = re.sub(r'http\S+', '', text)
    text = re.sub(r'<.*?>', '', text)
    return text.strip()


def build_full_text(fields: dict) -> str:
    text_defaults  = {k: "missing"  for k in ["title", "description", "requirements",
                                               "company_profile", "benefits"]}
    other_defaults = {k: "unknown"  for k in ["industry", "function"]}
    defaults = {**text_defaults, **other_defaults}

    def get(key):
        val = fields.get(key, "").strip()
        return clean_text(val) if val else defaults[key]

    return (
        f"TITLE: {get('title')} "
        f"DESC: {get('description')} "
        f"REQ: {get('requirements')} "
        f"COMPANY: {get('company_profile')} "
        f"BENEFITS: {get('benefits')} "
        f"INDUSTRY: {get('industry')} "
        f"FUNCTION: {get('function')}"
    )


def predict(full_text: str):
    inputs = tokenizer(
        full_text,
        return_tensors="pt",
        truncation=True,
        padding=True,
        max_length=512,
    )
    with torch.no_grad():
        logits = model(**inputs).logits
    probs = torch.softmax(logits, dim=1)[0]
    label = torch.argmax(logits, dim=1).item()
    return label, probs[1].item()
